### Task 1 — cron builder utility

**File:** `apps/api/src/modules/hr/hr-shared/cron-builder.util.ts`

```ts
import type { ScheduleConfig } from '../hr-task-template/hr-task-template.schema.js';

/**
 * Convert HrTaskTemplate.scheduleConfig into a node-cron expression.
 *
 * Weekly (ISO days 1=Mon..7=Sun) → "M H * * D,D,D" (cron 0=Sun..6=Sat).
 * Monthly (dayOfMonth 1..31)      → "M H D * *".
 *
 * Examples:
 *   {time:"09:00", mode:"weekly",  days:[1,2,3,4,5]} → "0 9 * * 1,2,3,4,5"
 *   {time:"23:30", mode:"weekly",  days:[7]}         → "30 23 * * 0"        (Sunday)
 *   {time:"15:00", mode:"monthly", dayOfMonth:15}    → "0 15 15 * *"
 */
export function buildCronExpr(cfg: ScheduleConfig): string {
  const [hStr, mStr] = cfg.time.split(':');
  const hour = Number(hStr);
  const minute = Number(mStr);
  if (!Number.isFinite(hour) || !Number.isFinite(minute)) {
    throw new Error(`Noto'g'ri vaqt formati: ${cfg.time}`);
  }
  if (cfg.mode === 'weekly') {
    if (!cfg.days?.length) throw new Error('weekly mode uchun days[] kerak');
    // ISO 1..7 (Mon..Sun) → cron 1..6,0 (Mon..Sat,Sun)
    const cronDays = cfg.days.map((d) => (d === 7 ? 0 : d)).sort((a, b) => a - b);
    return `${minute} ${hour} * * ${cronDays.join(',')}`;
  }
  if (cfg.mode === 'monthly') {
    if (cfg.dayOfMonth === undefined) throw new Error('monthly mode uchun dayOfMonth kerak');
    return `${minute} ${hour} ${cfg.dayOfMonth} * *`;
  }
  throw new Error(`Noto'g'ri mode: ${(cfg as { mode: string }).mode}`);
}
```

```ts
import { Inject, Injectable, Logger, type OnModuleInit } from '@nestjs/common';
import { SchedulerRegistry } from '@nestjs/schedule';
import { CronJob } from 'cron';
import { PrismaService } from '../../../prisma/prisma.service.js';
import { HR_TZ } from '../hr-shared/tz.util.js';
import { buildCronExpr } from '../hr-shared/cron-builder.util.js';
import { HrTaskSendService } from '../hr-task-send/hr-task-send.service.js';
import type { ScheduleConfig } from '../hr-task-template/hr-task-template.schema.js';

@Injectable()
export class HrTemplateSchedulerService implements OnModuleInit {
  private readonly logger = new Logger(HrTemplateSchedulerService.name);

  constructor(
    @Inject(PrismaService) private readonly prisma: PrismaService,
    @Inject(SchedulerRegistry) private readonly registry: SchedulerRegistry,
    @Inject(HrTaskSendService) private readonly sendService: HrTaskSendService,
  ) {}

  /** Scan DB and register all active scheduled templates on boot. */
  async onModuleInit() {
    const active = await this.prisma.client.hrTaskTemplate.findMany({
      where: { triggerType: 'scheduled', isActive: true },
      select: { id: true, accountId: true, scheduleConfig: true },
    });
    let registered = 0;
    for (const tpl of active) {
      try {
        this.register(tpl.id, tpl.accountId, tpl.scheduleConfig as unknown as ScheduleConfig);
        registered++;
      } catch (e) {
        this.logger.error(`Failed to register template ${tpl.id}: ${(e as Error).message}`);
      }
    }
    this.logger.log(`Registered ${registered}/${active.length} scheduled HR templates`);
  }

  /** Idempotent: removes existing job for this id then re-registers. */
  register(templateId: string, accountId: string, cfg: ScheduleConfig | null): void {
    this.unregister(templateId);
    if (!cfg) return;
    const expr = buildCronExpr(cfg);
    const job = new CronJob(
      expr,
      () => {
        void this.sendService.dispatchTemplate(accountId, templateId, {
          triggeredBy: 'scheduled',
        });
      },
      null,
      false,
      HR_TZ,
    );
    this.registry.addCronJob(this.jobName(templateId), job);
    job.start();
  }

  unregister(templateId: string): void {
    const name = this.jobName(templateId);
    if (this.registry.doesExist('cron', name)) {
      this.registry.deleteCronJob(name);
    }
  }

  private jobName(templateId: string): string {
    return `hr-template-${templateId}`;
  }
}
```

**Tests** (6):
- onModuleInit registers N active scheduled templates from DB
- register adds CronJob with correct expr (mock SchedulerRegistry)
- register is idempotent (calling twice does not throw, second replaces first)
- register with null config is no-op
- unregister removes existing job; no-op if absent
- onModuleInit continues past one failed template (logs error)

```ts
export const DispatchTemplateSchema = z.object({
  templateId: z.string().uuid(),

  employeeId: z.string().uuid().optional(),
});

  .object({
    type: z.enum(['yes', 'no', 'text']),
    text: z.string().max(2000).optional().nullable(),
  })
  .refine((v) => (v.type === 'text' ? !!v.text && v.text.trim().length > 0 : true), {
    path: ['text'],
    message: "'text' javob uchun matn kiritilishi shart",

  'answered_yes',
  'answered_no',

  'approved',
```

### Task 4 — HrTaskSendService

**File:** `apps/api/src/modules/hr/hr-task-send/hr-task-send.service.ts`

```ts
import {
  BadRequestException,
  ForbiddenException,
  Inject,
  Injectable,
  Logger,
  NotFoundException,
} from '@nestjs/common';
import { EventEmitter2 } from '@nestjs/event-emitter';
import { PrismaService } from '../../../prisma/prisma.service.js';
import { HR_EVENT } from '../hr-shared/hr-events.types.ts';   // see existing file
import type {
  AnswerTaskInput,
  DispatchTemplateInput,
  HrTaskLogStatus,
  ListLogsFilter,
} from './hr-task-send.schema.js';

interface DispatchOptions {
  triggeredBy: 'manual' | 'scheduled' | 'event';
  /** Override default assignee (manual re-dispatch). */
  employeeIdOverride?: string;
  sourceEventType?: string;
  sourceDocId?: string;
}

@Injectable()
export class HrTaskSendService {
  private readonly logger = new Logger(HrTaskSendService.name);

  constructor(
    @Inject(PrismaService) private readonly prisma: PrismaService,
    @Inject(EventEmitter2) private readonly events: EventEmitter2,
  ) {}

  /** Public manual dispatch (admin POSTs /hr/tasks/send). */
  async dispatch(accountId: string, input: DispatchTemplateInput) {
    return this.dispatchTemplate(accountId, input.templateId, {
      triggeredBy: 'manual',
      employeeIdOverride: input.employeeId,
    });
  }

  /** Core dispatch — used by manual, scheduled cron, and (future) event listeners. */
  async dispatchTemplate(accountId: string, templateId: string, opts: DispatchOptions) {
    const tpl = await this.prisma.client.hrTaskTemplate.findFirst({
      where: { id: templateId, accountId, isActive: true },
    });
    if (!tpl) throw new NotFoundException("Faol shablon topilmadi");

    const employeeId = await this.resolveAssignee(accountId, tpl, opts.employeeIdOverride);

    // Atomic: create Task + HrTaskLog + (optional) HrTelegramOutbox
    return this.prisma.client.$transaction(async (tx) => {
      const task = await tx.task.create({
        data: {
          accountId,
          assigneeId: employeeId,
          title: tpl.title,
          description: tpl.description,
          status: 'open',
          priority: this.mapPriority(tpl.priority),
          kind: 'HR',
          hrTemplateId: tpl.id,
          hrCheckerId: tpl.checkerId,
          hrResponseType: tpl.responseType,
          hrDeadlineMinutes: tpl.deadlineMinutes,
          hrRewardMinor: tpl.rewardMinor,
          hrFineMinor: tpl.fineMinor,
          hrDependsOnId: tpl.dependsOnId,
        },
      });

      const log = await tx.hrTaskLog.create({
        data: {
          accountId,
          templateId: tpl.id,
          taskId: task.id,
          employeeId,
          status: 'sent',
          sentAt: new Date(),
        },
      });

      // Enqueue Telegram outbox if employee has phone (worker comes in P4).
      const emp = await tx.employee.findUnique({
        where: { id: employeeId },
        select: { telegramPhone: true },
      });
      if (emp?.telegramPhone) {
        await tx.hrTelegramOutbox.create({
          data: {
            accountId,
            employeeId,
            toPhone: emp.telegramPhone,
            messageText: this.renderTaskMessage(tpl.title, tpl.description),
            status: 'pending',
            sourceEventType: `hr.task.${opts.triggeredBy}`,
            sourceDocId: log.id,
          },
        });
      }

      return { task, log, telegramQueued: !!emp?.telegramPhone };
    });
  }

  /** Xodim javob beradi. Anti-self-approval applies on review side, not here. */
  async recordAnswer(
    accountId: string,
    taskLogId: string,
    actingEmployeeId: string,
    input: AnswerTaskInput,
  ) {
    const log = await this.prisma.client.hrTaskLog.findFirst({
      where: { id: taskLogId, accountId },
      include: { template: true },
    });
    if (!log) throw new NotFoundException('Vazifa topilmadi');
    if (log.employeeId !== actingEmployeeId) {
      throw new ForbiddenException('Bu vazifa sizniki emas');
    }
    if (log.status !== 'sent') {
      throw new BadRequestException(`Bu vazifaga allaqachon javob berilgan (${log.status})`);
    }

    const hasChecker = !!log.template.checkerId;
    const requiresReview = hasChecker && (input.type === 'yes' || input.type === 'text');
    const newStatus: HrTaskLogStatus = requiresReview
      ? 'pending_review'
      : input.type === 'yes'
        ? 'answered_yes'
        : input.type === 'no'
          ? 'answered_no'
          : 'answered_text';

    const updated = await this.prisma.client.hrTaskLog.update({
      where: { id: log.id },
      data: {
        status: newStatus,
        responseText: input.text ?? null,
        answeredAt: new Date(),
      },
    });

    // Finalize side-effects (only when no further review needed).
    if (!requiresReview) {
      await this.finalize(accountId, updated.id, newStatus, log.templateId, log.employeeId, null);
    }
    return updated;
  }

  /**
   * Finalize callbacks — bonus/fine + depends_on chain + domain event.
   * Called from recordAnswer (no checker path) AND from review service in P3c.
   */
  async finalize(
    accountId: string,
    taskLogId: string,
    status: 'answered_yes' | 'answered_no' | 'answered_text' | 'approved' | 'rejected',
    templateId: string,
    employeeId: string,
    reviewedById: string | null,
  ) {
    // 1. auto bonus/fine
    const tpl = await this.prisma.client.hrTaskTemplate.findUnique({
      where: { id: templateId },
    });
    if (tpl) {
      const isReward =
        status === 'answered_yes' || status === 'answered_text' || status === 'approved';
      const isFine = status === 'answered_no' || status === 'rejected';
      if (isReward && tpl.rewardMinor && tpl.rewardMinor > 0n) {
        await this.prisma.client.hrBonusFineLog.create({
          data: {
            accountId,
            employeeId,
            kind: 'bonus',
            source: 'auto_task_reward',
            amountMinor: tpl.rewardMinor,
            reason: `Vazifa: ${tpl.title}`,
            taskLogId,
            createdById: null,
          },
        });
      }
      if (isFine && tpl.fineMinor && tpl.fineMinor > 0n) {
        await this.prisma.client.hrBonusFineLog.create({
          data: {
            accountId,
            employeeId,
            kind: 'fine',
            source: 'auto_task_fine',
            amountMinor: tpl.fineMinor,
            reason: `Vazifa rad etildi: ${tpl.title}`,
            taskLogId,
            createdById: null,
          },
        });
      }
    }

    // 2. depends_on chain — dispatch dependents for this employee
    if (status === 'answered_yes' || status === 'answered_text' || status === 'approved') {
      const dependents = await this.prisma.client.hrTaskTemplate.findMany({
        where: { dependsOnId: templateId, isActive: true, accountId },
      });
      for (const dep of dependents) {
        try {
          await this.dispatchTemplate(accountId, dep.id, {
            triggeredBy: 'event',
            employeeIdOverride: employeeId, // same employee for chain
            sourceEventType: 'hr.task.chain',
            sourceDocId: taskLogId,
          });
        } catch (e) {
          this.logger.warn(
            `Dependent dispatch failed (tpl=${dep.id}): ${(e as Error).message}`,
          );
        }
      }
    }

    // 3. domain event
    this.events.emit(HR_EVENT.HR_TASK_LOG_FINALIZED, {
      accountId,
      taskLogId,
      templateId,
      employeeId,
      status,
      reviewedById,
    });
  }

  async listLogs(accountId: string, scopeEmployeeId: string | null, filter: ListLogsFilter) {
    const where: Record<string, unknown> = { accountId };
    if (scopeEmployeeId) where.employeeId = scopeEmployeeId; // own_only enforcement
    if (filter.templateId) where.templateId = filter.templateId;
    if (filter.employeeId) where.employeeId = filter.employeeId;
    if (filter.status) where.status = filter.status;
    if (filter.dateFrom || filter.dateTo) {
      where.sentAt = {
        ...(filter.dateFrom && { gte: filter.dateFrom }),
        ...(filter.dateTo && { lte: filter.dateTo }),
      };
    }
    const [rows, total] = await this.prisma.client.$transaction([
      this.prisma.client.hrTaskLog.findMany({
        where,
        orderBy: { sentAt: 'desc' },
        skip: (filter.page - 1) * filter.limit,
        take: filter.limit,
        include: {
          template: { select: { id: true, title: true, responseType: true } },
          employee: { select: { id: true, name: true } },
          reviewedBy: { select: { id: true, name: true } },
        },
      }),
      this.prisma.client.hrTaskLog.count({ where }),
    ]);
    return { rows, total, page: filter.page, limit: filter.limit };
  }

  // ─── private helpers ────────────────────────────────────────────────────

  private async resolveAssignee(
    accountId: string,
    tpl: { assignedEmployeeId: string | null; assignedRole: string | null },
    override: string | undefined,
  ): Promise<string> {
    if (override) {
      const emp = await this.prisma.client.employee.findFirst({
        where: { id: override, accountId, archived: false },
        select: { id: true },
      });
      if (!emp) throw new BadRequestException('Override xodim topilmadi');
      return emp.id;
    }
    if (tpl.assignedEmployeeId) return tpl.assignedEmployeeId;
    if (tpl.assignedRole) {
      // Yangibolim: pick first employee with this role (deterministic by createdAt).
      const candidate = await this.prisma.client.employee.findFirst({
        where: { accountId, archived: false, hrRoles: { has: tpl.assignedRole } },
        orderBy: { createdAt: 'asc' },
        select: { id: true },
      });
      if (!candidate) {
        throw new BadRequestException(`'${tpl.assignedRole}' rolida xodim topilmadi`);
      }
      return candidate.id;
    }
    throw new BadRequestException('Shablonda assignee belgilanmagan');
  }

  private mapPriority(hrPriority: string): string {
    // hr_task_template uses {low, medium, high, urgent}; Task model uses {low, normal, high, urgent}
    return hrPriority === 'medium' ? 'normal' : hrPriority;
  }

  private renderTaskMessage(title: string, description: string | null): string {
    // P4 ga qadar oddiy text; keyin Eta template engine.
    return description ? `📋 ${title}\n\n${description}` : `📋 ${title}`;
  }
}
```

```ts
import { Body, Controller, Get, Inject, Param, Post, Query, UseGuards } from '@nestjs/common';

  async answer(
    @CurrentUser() user: AuthenticatedUser,
    @Param('id') id: string,
    @Body() body: unknown,
  ) {
    const input = AnswerTaskSchema.parse(body);
    return this.svc.recordAnswer(user.accountId, id, user.sub, input);
  }

  /** List logs (own_only filter — admin sees all, xodim only theirs). */
  @Get('logs')
  @RequireHrPermission('tasks', 'read')
  async list(@CurrentUser() user: AuthenticatedUser, @Query() query: unknown) {
    const filter = ListLogsFilterSchema.parse(query);
    const isAdmin = user.hrRoles?.includes('admin') ?? false;
    return this.svc.listLogs(user.accountId, isAdmin ? null : user.sub, filter);
  }
}
```

| Risk | Mitigation in this sprint |
|---|---|
| **Race: 2 admin dispatchTemplate same template same employee** | Allowed — duplicate tasks acceptable per yangibolim parity. If unique constraint needed → P3c review queue dedup |
| **Race: scheduled cron fires while admin manually dispatches** | Allowed (same as above). |

| **Anti-self-approval** | Enforced at template create (Zod superRefine) AND at review service (P3c) |
| **Scheduler boot fail one bad cron** | try/catch per template, log + continue |
| **Outbox enqueue without phone** | Skipped silently (admin sees Task but not message — UI shows badge in P3d) |
| **`null` schedule config on registration** | Defensive no-op |
| **BigInt money** | rewardMinor/fineMinor stay BigInt throughout, never cast to Number |
| **Transaction rollback** | Task + Log + Outbox in single `$transaction` — atomic |
